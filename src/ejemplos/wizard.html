Indicador con diez pasos

    
Reduce el ancho de la ventana, desplaza el indicador o usa sus flechas. Los botones inferiores cambian de paso.

    

    
import { Wizard } from '../js/boundary/componentes/navigation/wizard/index.js';
const wizard = new Wizard('#patientWizard', {
    stepCarousel: { enabled: true, autoScroll: true, showArrows: true, smoothScroll: true },
    steps: [
        { id: 'personal', title: 'Datos personales', content: personalForm,
          validate: () => personalForm.checkValidity() },
        { id: 'contact', title: 'Contacto', content: contactForm,
          validate: async () => await validateContactData() },
        { id: 'review', title: 'Revisar', content: confirmationElement }
    ],
    onComplete: async data => { /* Acción proporcionada por la página */ }
});
wizard.render();
